# មេរៀនទី ២៖ អថេរ ការបញ្ចូល និងការបង្ហាញទិន្នន័យ

## គោលដៅ

- រក្សាទុកតម្លៃក្នុងអថេរ និងដាក់ឈ្មោះត្រឹមត្រូវ។
- អានព័ត៌មានពីអ្នកប្រើដោយ <code>input()</code>។
- បង្ហាញលទ្ធផល និងបញ្ចូលតម្លៃក្នុងអត្ថបទ។

## ១. អថេររក្សាទុកតម្លៃ

~~~python
student_name = "ដារ៉ា"
lesson_count = 2
print(student_name)
print(lesson_count)
~~~

សញ្ញា <code>=</code> ក្នុងឧទាហរណ៍នេះមានន័យថា assign តម្លៃទៅអថេរ។ ឈ្មោះអថេរគួរពិពណ៌នាតួនាទី ហើយ Python ប្រើអក្សរតូចភ្ជាប់គ្នាដោយ underscore ដូចជា <code>total_price</code>។ ឈ្មោះអថេរមិនអាចចាប់ផ្តើមដោយលេខ ឬប្រើ keyword របស់ Python បានទេ។

## ២. ទទួល input

~~~python
name = input("តើអ្នកឈ្មោះអ្វី? ")
print("សួស្តី", name)
~~~

<code>input()</code> បង្ហាញសំណួរ រង់ចាំអ្នកវាយ ហើយត្រឡប់លទ្ធផលជាអក្សរ។ ទោះអ្នកវាយលេខក៏ដោយ វានៅតែជា string រហូតដល់បម្លែងប្រភេទ។

## ៣. បង្ហាញលទ្ធផលជាមួយ f-string

~~~python
name = "សុភា"
lessons = 3
print(f"{name} បានរៀន {lessons} មេរៀន។")
~~~

ដាក់អក្សរ <code>f</code> នៅមុខ string ហើយដាក់ឈ្មោះអថេរក្នុង <code>{}</code>។ វាអាចជួយអោយឃើញច្បាស់ជាងភ្ជាប់ string ដោយ <code>+</code>។

## ៤. បញ្ចូល និងគណនា

~~~python
first = int(input("លេខទីមួយ: "))
second = int(input("លេខទីពីរ: "))
print(f"ផលបូក = {first + second}")
~~~

យើងបម្លែង input ទៅជា integer មុនបូក។ ប្រសិនបើអ្នកប្រើវាយអក្សរមិនមែនលេខ ការបម្លែងនឹងបរាជ័យ។ មេរៀន exception នឹងបង្រៀនការគ្រប់គ្រងករណីនោះ។

## ជំហានសាកល្បង

1. រក្សាកូដក្នុង <code>greeting.py</code>។
2. រត់ <code>python greeting.py</code>។
3. វាយឈ្មោះ និងពិនិត្យថាវាត្រូវបានបង្ហាញត្រឡប់មកវិញ។
4. សាក input ទទេ ដើម្បីមើលថាកម្មវិធីប្រព្រឹត្តយ៉ាងដូចម្តេច។

## កំហុសទូទៅ

- សរសេរ <code>name == "Dara"</code> ដើម្បីកំណត់តម្លៃ។ សញ្ញា <code>==</code> ប្រើប្រៀបធៀប មិនមែន assign ទេ។
- ភ្លេចបម្លែង <code>input()</code> មុនគណនាលេខ។
- សរសេរអថេរមួយជា <code>student_name</code> ហើយហៅជា <code>studentName</code>។ Python ប្រកាន់អក្សរធំតូច។
